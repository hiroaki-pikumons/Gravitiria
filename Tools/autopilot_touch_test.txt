# 期待: Title でタップ → Tutorial へ。Tutorial で tlog の move=(0,1) → 前進、aim=True & timeScale=0.10、離して gravity が (1,0,0) に、
#       ポーズボタンで paused=True、再開ボタンで False。

[Title]
wait 2
tlog
tdown 1 0.5 0.5
wait 0.1

wait 3

[Tutorial]
touchui on
manual
tlog
# 仮想スティック: 左側に置いて上へ
tdown 1 0.2 0.3
wait 0.1
tmove 1 0.2 0.42
wait 0.2
tlog stick pushed
wait 1.3
tup 1
wait 0.3
tlog stick released
# 視点スワイプ（右側）

# 右の壁を狙って、重力ボタン長押し → 離す

tlog after release
